# Errors — RFC 9457 application/problem+json.
# Drop into components.

components:
  schemas:
    Problem:
      type: object
      title: Problem
      description: |
        A problem details object as defined by RFC 9457. Identifies and
        describes a problem that occurred during processing of a request.
      required: [type, title, status]
      properties:
        type:
          type: string
          format: uri
          description: |
            URI reference identifying the problem type. Should be
            dereferenceable to human-readable documentation.
            Use `about:blank` for problems with no specific type.
          examples:
            - https://errors.example.com/payment-declined
            - about:blank
        title:
          type: string
          description: |
            Short, human-readable summary of the problem. Same for
            all occurrences of the same `type`.
          examples: [Payment declined]
        status:
          type: integer
          minimum: 100
          maximum: 599
          description: HTTP status code generated by the origin server.
          examples: [402]
        detail:
          type: string
          description: |
            Human-readable explanation specific to this occurrence.
            Should not include implementation details that aid attackers.
        instance:
          type: string
          format: uri-reference
          description: |
            URI reference identifying the specific occurrence —
            typically the request URI or a correlation/trace identifier.
        code:
          type: string
          pattern: '^[a-z][a-z0-9_]*(\.[a-z][a-z0-9_]*)*$'
          description: |
            Stable application error code, dot-separated. Stable across
            versions for the same logical error.
          examples: [payment.declined, validation.required_field]
        errors:
          type: array
          description: |
            Field-level errors when this is a validation failure.
          items:
            type: object
            required: [field, code]
            properties:
              field:
                type: string
                description: JSON pointer or dotted path to the offending field.
                examples: ['/passengers/0/dateOfBirth', 'payment.cvv']
              code:
                type: string
                description: Field-level error code.
                examples: [required, invalid_format, out_of_range]
              message:
                type: string
                description: Human-readable field-level message.

  responses:
    BadRequest:
      description: The request was malformed or failed validation.
      content:
        application/problem+json:
          schema: { $ref: '#/components/schemas/Problem' }
          examples:
            validation: { $ref: '#/components/examples/ValidationProblem' }

    Unauthorized:
      description: Missing or invalid credentials.
      content:
        application/problem+json:
          schema: { $ref: '#/components/schemas/Problem' }

    Forbidden:
      description: Authenticated but not authorized for this operation.
      content:
        application/problem+json:
          schema: { $ref: '#/components/schemas/Problem' }

    NotFound:
      description: The resource does not exist or is hidden from this caller.
      content:
        application/problem+json:
          schema: { $ref: '#/components/schemas/Problem' }
          examples:
            notFound: { $ref: '#/components/examples/NotFoundProblem' }

    Conflict:
      description: |
        State conflict. Common causes: optimistic-lock failure (`If-Match`),
        idempotency-key reuse with a different body, illegal state transition.
      content:
        application/problem+json:
          schema: { $ref: '#/components/schemas/Problem' }

    PreconditionFailed:
      description: Optimistic concurrency check failed (`If-Match` mismatch).
      content:
        application/problem+json:
          schema: { $ref: '#/components/schemas/Problem' }

    UnprocessableEntity:
      description: Semantic validation failed; the request was well-formed but invalid.
      content:
        application/problem+json:
          schema: { $ref: '#/components/schemas/Problem' }

    TooManyRequests:
      description: Rate limit exceeded.
      headers:
        Retry-After:
          schema: { type: integer }
          description: Seconds to wait before retrying.
      content:
        application/problem+json:
          schema: { $ref: '#/components/schemas/Problem' }

    InternalError:
      description: Unexpected server error.
      content:
        application/problem+json:
          schema: { $ref: '#/components/schemas/Problem' }

  examples:
    ValidationProblem:
      summary: Field validation failure
      value:
        type: https://errors.example.com/validation

        status: 422
